use aien_runtime::requirements::analyze;
fn main() {
    let path = std::env::args().nth(1).unwrap();
    let v: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(path).unwrap()).unwrap();
    let arr = v.get("tasks").and_then(|t| t.as_array()).or_else(|| v.as_array()).unwrap().clone();
    for t in arr {
        let ex = analyze(t["goal"].as_str().unwrap());
        let s = if !ex.uncertain.is_empty() { "REFUSED" } else if ex.requirements.is_empty() { "SILENT" } else { "RECOGNIZED" };
        let labels: Vec<String> = ex.requirements.iter().map(|r| r.label()).collect();
        println!("{}\t{}\t{:?}\t{:?}", t["id"].as_str().unwrap(), s, ex.uncertain, labels);
    }
}
