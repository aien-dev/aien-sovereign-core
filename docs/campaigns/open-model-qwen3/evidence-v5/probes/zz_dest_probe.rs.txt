// Throwaway probe (not committed as code): what the product decides for each goal's destination, in a workspace
// holding the task's seed files (as NP1_SEED_DIR does), via spine::classify_destination (the single decision of a task).
use aien_runtime::spine::{classify_destination, TargetClass};
use std::path::Path;
fn copy(src: &Path, dst: &Path) {
    for e in std::fs::read_dir(src).unwrap() {
        let e = e.unwrap();
        let to = dst.join(e.file_name());
        if e.file_type().unwrap().is_dir() {
            std::fs::create_dir_all(&to).unwrap();
            copy(&e.path(), &to);
        } else {
            std::fs::copy(e.path(), &to).unwrap();
        }
    }
}
fn main() {
    let tasks = std::env::args().nth(1).unwrap();
    let base = Path::new(&tasks).parent().unwrap().to_path_buf();
    let scratch = Path::new(&std::env::args().nth(2).unwrap()).to_path_buf();
    let v: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(&tasks).unwrap()).unwrap();
    for t in v["tasks"].as_array().unwrap() {
        let id = t["id"].as_str().unwrap();
        let ws = scratch.join(id).join("ws");
        std::fs::create_dir_all(&ws).unwrap();
        if let Some(s) = t["seed"].as_str() {
            copy(&base.join(s), &ws);
        }
        let (class, dest) = classify_destination(t["goal"].as_str().unwrap(), &ws);
        let c = match class {
            TargetClass::New => "NEW".to_string(),
            TargetClass::Edit(p, _) => format!("EDIT {p}"),
            TargetClass::Refused(why) => format!("REFUSED {why}"),
        };
        println!("{id}\t{}\t{:?}\t{c}", t["destination"].as_str().unwrap_or("null"), dest);
    }
}
