use aien_runtime::requirements::Requirement;
fn main() {
    // (topic as the goal names it, a sentence using a form the goal itself accepts)
    let cases = [
        ("drying", "Leave the pods until they have dried on the plant."),
        ("drying", "Dry the seeds on a plate."),
        ("labelling", "Label each envelope with the variety."),
        ("labelling", "Keep every packet labelled."),
        ("storage", "Store the jars somewhere cool."),
        ("storage", "The seeds are stored in a tin."),
        ("withdraw", "She withdrew fifty pounds."),
        ("withdraw", "Withdrawing money is free."),
        ("interest", "The bank pays interest monthly."),
        ("balance", "Check the balances each week."),
    ];
    for (topic, text) in cases {
        let r = Requirement::RequiredTopics(vec![topic.to_string()]);
        let out = r.check(text).map_or("MET".to_string(), |why| format!("UNMET ({why})"));
        println!("{topic}\t{text}\t{out}");
    }
}
