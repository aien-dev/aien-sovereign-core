use aien_runtime::requirements::{analyze, Requirement};
fn main() {
    // G2: the D3 topics exactly as the product reads them from the goal, checked on one sentence per accepted form
    // of the task file (tasks-oq3-v5.json, requirement kind topic, field accepted).
    let path = std::env::args().nth(1).unwrap();
    let v: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(path).unwrap()).unwrap();
    let d3 = v["tasks"].as_array().unwrap().iter().find(|t| t["id"] == "D3").unwrap().clone();
    let ex = analyze(d3["goal"].as_str().unwrap());
    println!("# uncertain spans: {:?}", ex.uncertain);
    let topics: Vec<String> = ex.requirements.iter().filter_map(|r| match r {
        Requirement::RequiredTopics(t) => Some(t.clone()),
        _ => None,
    }).flatten().collect();
    println!("# topics as read: {:?}", topics);
    for req in d3["requirements"].as_array().unwrap().iter().filter(|r| r["kind"] == "topic") {
        let name = req["name"].as_str().unwrap();
        let read: Vec<&String> = topics.iter().filter(|t| t.split_whitespace().next() == Some(name)).collect();
        if read.len() != 1 { println!("{name}\t-\t-\tNOT READ ONCE ({} matches)", read.len()); continue; }
        for form in req["accepted"].as_array().unwrap() {
            let text = format!("This part is about {} today.", form.as_str().unwrap());
            let r = Requirement::RequiredTopics(vec![read[0].clone()]);
            let out = r.check(&text).map_or("MET".to_string(), |why| format!("UNMET ({why})"));
            println!("{name}\t{}\t{text}\t{out}", read[0]);
        }
    }
}
